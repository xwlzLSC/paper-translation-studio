cmake_minimum_required(VERSION 3.22)
project(PaperPrecision LANGUAGES C CXX)
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned llama.cpp v0.9.0 source directory.")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)

add_library(paper_precision SHARED precision.cpp)
target_compile_features(paper_precision PRIVATE cxx_std_17)
target_link_libraries(paper_precision PRIVATE llama)
if(ANDROID)
    target_link_options(paper_precision PRIVATE "-Wl,--exclude-libs,ALL")
else()
    find_package(JNI REQUIRED)
    target_include_directories(paper_precision PRIVATE ${JNI_INCLUDE_DIRS})
endif()
